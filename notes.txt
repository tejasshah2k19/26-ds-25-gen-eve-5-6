boolean findTarget(int nums[], int target){

    for(int i=0;i<nums.length;i++){
        for(int j=i+1;j<nums.length;j++){
            int sum = nums[i] + nums[j]; 
            if(sum == target){
                return true;
            }
        }
    }
    return false;

}
